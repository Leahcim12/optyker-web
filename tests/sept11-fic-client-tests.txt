// OPTYKER_SEPT11_CLIENT_BILLING_TESTS -- mocked provider only.
test('customer association survives preview and confirmed provider creation',async()=>{const {db}=await setup();const cid=crypto.randomUUID();db.tables.optyker_clients=[{id:cid}];const p=await issuance(db,'fic_preview',{...base,client_id:cid});assert.equal(p.client_id,cid);await issuance(db,'fic_create',{id:p.id,confirm_create:true,confirm_date:true});assert.equal(db.tables.optyker_billing_invoices[0].client_id,cid);});
test('draft cannot silently switch its linked customer',async()=>{const {db,remote}=await setup();const a=crypto.randomUUID(),b=crypto.randomUUID();db.tables.optyker_clients=[{id:a},{id:b}];const p=await issuance(db,'fic_preview',{...base,client_id:a});await assert.rejects(issuance(db,'fic_preview',{...base,client_id:b,draft_id:p.id}),/collegato/);assert.equal(remote.length,0);});
test('POS request reuses original invoice row and remains idempotent',async()=>{const {db,remote}=await setup();const cid=crypto.randomUUID(),pid=crypto.randomUUID();db.tables.optyker_clients=[{id:cid}];db.tables.optyker_billing_invoices.push({id:pid,client_id:cid,direction:'outgoing',currency:'EUR',sdi_status:'draft',total:122,provider_invoice_id:null,provider_payload:{source:'optyker_pos'}});const p=await issuance(db,'fic_preview',{...base,client_id:cid,pos_source_id:pid});assert.equal(p.pos_source_id,pid);await issuance(db,'fic_create',{id:p.id,confirm_create:true,confirm_date:true});await issuance(db,'fic_create',{id:p.id,confirm_create:true,confirm_date:true});assert.equal(remote.length,1);assert.equal(db.tables.optyker_billing_invoices.length,1);assert.equal(db.tables.optyker_billing_invoices[0].id,pid);assert.match(db.tables.optyker_billing_invoices[0].provider_invoice_id,/^fic:/);});
test('POS amount mismatch cannot produce an invoice',async()=>{const {db,remote}=await setup();const cid=crypto.randomUUID(),pid=crypto.randomUUID();db.tables.optyker_clients=[{id:cid}];db.tables.optyker_billing_invoices.push({id:pid,client_id:cid,direction:'outgoing',currency:'EUR',sdi_status:'draft',total:80,provider_invoice_id:null,provider_payload:{source:'optyker_pos'}});await assert.rejects(issuance(db,'fic_preview',{...base,client_id:cid,pos_source_id:pid}),/coincidere/);assert.equal(remote.length,0);assert.equal(db.tables.optyker_fic_drafts.length,0);});
test('unknown customer and mismatched POS owner are rejected',async()=>{const {db}=await setup();const cid=crypto.randomUUID(),other=crypto.randomUUID(),pid=crypto.randomUUID();db.tables.optyker_clients=[{id:cid}];await assert.rejects(issuance(db,'fic_preview',{...base,client_id:other}),/Cliente/);db.tables.optyker_billing_invoices.push({id:pid,client_id:other,direction:'outgoing',currency:'EUR',sdi_status:'draft',total:122,provider_invoice_id:null,provider_payload:{source:'optyker_pos'}});await assert.rejects(issuance(db,'fic_preview',{...base,client_id:cid,pos_source_id:pid}),/questo cliente/);});
